cmake_minimum_required(VERSION 3.24)
project(headless_csgo VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32)
  message(FATAL_ERROR "This first release targets Windows.")
endif()
add_library(hc_core STATIC src/proto.cpp src/wire.cpp src/netchannel.cpp src/state.cpp src/usercmd.cpp src/auth.cpp src/session.cpp src/hash.cpp)
target_include_directories(hc_core PUBLIC include third_party)
target_compile_definitions(hc_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(hc_core PUBLIC ws2_32 bcrypt)
target_compile_options(hc_core PRIVATE /W4 /permissive- /utf-8)
add_executable(headless-csgo src/main.cpp)
target_link_libraries(headless-csgo PRIVATE hc_core)
set(STEAM_SDK_DIR "E:/archives/csgo/csgo_partner/src" CACHE PATH "Steam SDK headers and import libraries")
set(STEAM_API_DLL "D:/SteamLibrary/steamapps/common/Counter-Strike Global Offensive/bin/steam_api.dll" CACHE FILEPATH "Permitted Steamworks SDK authentication redistributable")
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Configure with -A Win32 for the local Steam SDK import library.")
endif()
add_executable(steam-auth-helper src/steam_auth_helper.cpp)
set_target_properties(steam-auth-helper PROPERTIES OUTPUT_NAME steam_auth_helper)
target_include_directories(steam-auth-helper PRIVATE "${STEAM_SDK_DIR}/public" third_party)
target_compile_definitions(steam-auth-helper PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(steam-auth-helper PRIVATE /W4 /utf-8)
target_link_libraries(steam-auth-helper PRIVATE "${STEAM_SDK_DIR}/lib/public/steam_api.lib")
add_custom_command(TARGET steam-auth-helper POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${STEAM_API_DLL}" "$<TARGET_FILE_DIR:steam-auth-helper>/steam_api.dll")
file(GENERATE OUTPUT "$<TARGET_FILE_DIR:steam-auth-helper>/steam_appid.txt" CONTENT "730\n")
include(CTest)
if(BUILD_TESTING)
  foreach(name wire state proto session)
    add_executable(${name}-tests tests/${name}_tests.cpp)
    target_link_libraries(${name}-tests PRIVATE hc_core)
    add_test(NAME ${name} COMMAND ${name}-tests)
  endforeach()
endif()
