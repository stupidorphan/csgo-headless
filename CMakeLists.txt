cmake_minimum_required(VERSION 3.25)
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT "$<$<CONFIG:Debug,RelWithDebInfo>:Embedded>")
project(headless_csgo VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32)
  message(FATAL_ERROR "This first release targets Windows.")
endif()
add_compile_options(/Brepro)
add_link_options(/Brepro /INCREMENTAL:NO)
add_library(hc_core STATIC src/proto.cpp src/wire.cpp src/netchannel.cpp src/state.cpp src/usercmd.cpp src/auth.cpp src/steam_cm.cpp src/qr.cpp third_party/qrcodegen/qrcodegen.cpp src/session.cpp src/hash.cpp)
target_include_directories(hc_core PUBLIC include third_party)
target_compile_definitions(hc_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(hc_core PUBLIC ws2_32 bcrypt winhttp crypt32 advapi32)
target_compile_options(hc_core PRIVATE /W4 /permissive- /utf-8)
add_executable(headless-csgo src/main.cpp)
target_link_libraries(headless-csgo PRIVATE hc_core)
file(READ "${CMAKE_SOURCE_DIR}/dependencies.lock.json" DEPENDENCY_LOCK)
function(require_pinned_file ROOT GROUP MEMBER INDEX)
  if(INDEX STREQUAL "")
    string(JSON PATH_VALUE GET "${DEPENDENCY_LOCK}" components ${GROUP} ${MEMBER} path)
    string(JSON EXPECTED GET "${DEPENDENCY_LOCK}" components ${GROUP} ${MEMBER} sha256)
  else()
    string(JSON PATH_VALUE GET "${DEPENDENCY_LOCK}" components ${GROUP} ${MEMBER} ${INDEX} path)
    string(JSON EXPECTED GET "${DEPENDENCY_LOCK}" components ${GROUP} ${MEMBER} ${INDEX} sha256)
  endif()
  file(SHA256 "${ROOT}/${PATH_VALUE}" ACTUAL)
  if(NOT ACTUAL STREQUAL EXPECTED)
    message(FATAL_ERROR "Pinned dependency hash mismatch: ${ROOT}/${PATH_VALUE}")
  endif()
endfunction()
string(JSON JSON_FILES LENGTH "${DEPENDENCY_LOCK}" components nlohmann files)
math(EXPR JSON_LAST "${JSON_FILES}-1")
foreach(INDEX RANGE ${JSON_LAST})
  require_pinned_file("${CMAKE_SOURCE_DIR}" nlohmann files ${INDEX})
endforeach()
string(JSON QR_FILES LENGTH "${DEPENDENCY_LOCK}" components qrcodegen files)
math(EXPR QR_LAST "${QR_FILES}-1")
foreach(INDEX RANGE ${QR_LAST})
  require_pinned_file("${CMAKE_SOURCE_DIR}" qrcodegen files ${INDEX})
endforeach()
